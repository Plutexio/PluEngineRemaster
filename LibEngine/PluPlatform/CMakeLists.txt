# L1 — window, OS event pumping, input backends.
plu_add_engine_module(PluPlatform DEPENDS PluCore)

target_link_libraries(PluPlatform PUBLIC glad imgui::imgui)
if (NOT WIN32)
    target_link_libraries(PluPlatform PRIVATE SDL3::SDL3)

    # The SDL window icon, compiled in as PNG bytes so it never depends on a file next to the
    # executable. Windows takes the icon from the executable's resources instead (Packaging/pluengine.rc).
    set(PLU_APP_ICON_PNG "${CMAKE_SOURCE_DIR}/Packaging/pluengine.png")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PLU_APP_ICON_PNG}")
    file(READ "${PLU_APP_ICON_PNG}" PLU_APP_ICON_HEX HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," PLU_APP_ICON_BYTES "${PLU_APP_ICON_HEX}")
    set(PLU_APP_ICON_DIR "${CMAKE_CURRENT_BINARY_DIR}/Generated")
    # file(CONFIGURE) rewrites the header only when its content changed, so a reconfigure does not
    # recompile SdlWindow.cpp for nothing.
    file(CONFIGURE OUTPUT "${PLU_APP_ICON_DIR}/AppIconPng.h" CONTENT
"// Generated by LibEngine/PluPlatform/CMakeLists.txt from Packaging/pluengine.png - do not edit.
#pragma once

namespace Plu::AppIcon
{
    inline constexpr unsigned char kPng[] = { @PLU_APP_ICON_BYTES@ };
}
" @ONLY)
    target_include_directories(PluPlatform PRIVATE "${PLU_APP_ICON_DIR}")
else ()
    target_link_libraries(PluPlatform PRIVATE OpenGL::GL dwmapi ole32 shell32 uuid)  # OpenGL::GL: wgl*
endif ()
